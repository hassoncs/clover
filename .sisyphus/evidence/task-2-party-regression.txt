**Location**: `api/src/party/PartyRoomDO.ts` (lines 590-608)

```typescript
async sendToPlayer(
  playerId: string,
  data: Record<string, unknown>,
): Promise<void> {
  const message = encodeMessage(privateStateMessage(data));

  for (const ws of this.sockets) {
    const meta = this.socketMetadata.get(ws);
    if (
      meta?.role === "player" &&
      meta.playerId === playerId &&
      ws.readyState === WebSocket.OPEN
    ) {
      try {
        ws.send(message);
      } catch {}
    }
  }
}
```

- ✅ Sends private_state only to targeted player
- ✅ Gracefully no-ops when player is not connected
